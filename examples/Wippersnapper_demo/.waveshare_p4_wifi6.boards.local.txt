# Waveshare ESP32-P4-WIFI6 stick builds with the generic "esp32p4" (ESP32P4 Dev Module)
# board definition, which only yields ARDUINO_ESP32P4_DEV (reserved for the Adafruit
# Metro ESP32-P4). Add the WipperSnapper board define so Wippersnapper_Boards.h picks
# the Waveshare entry. Installed by ci-arduino build_platform.py --boards-local-txt
# for this example when building the waveshare_p4_wifi6 platform only.
#
# NOTE: the PSRAM menu options set build.defines after the board-level value, so a
# plain esp32p4.build.defines override is clobbered; override the menu entries instead
# (PSRAM=disabled is the default when the FQBN does not select it).
esp32p4.menu.PSRAM.disabled.build.defines=-DWS_WAVESHARE_ESP32P4_WIFI6_STICK
esp32p4.menu.PSRAM.enabled.build.defines=-DBOARD_HAS_PSRAM -DWS_WAVESHARE_ESP32P4_WIFI6_STICK
