244 
245  if (strlen(_ssid) == 0) {
246  _status = WS_SSID_INVALID;
247  } else {

252  // ESP8266 MUST be in STA mode to avoid device acting as client/server
253  WiFi.mode(WIFI_STA);
254  _status = WS_NET_DISCONNECTED;
255  delay(100);
256 
257  if (WS._isWiFiMulti) {
258  // multi network mode
259  for (int i = 0; i < WS_MAX_ALT_WIFI_NETWORKS; i++) {
260  if (strlen(WS._multiNetworks[i].ssid) > 0 &&
261  (_wifiMulti.existsAP(WS._multiNetworks[i].ssid) == false)) {
262  // doesn't exist, add it
263  _wifiMulti.addAP(WS._multiNetworks[i].ssid,
264  WS._multiNetworks[i].pass);
265  }
266  }
267  }
268 
269  // add default network
270  if (_wifiMulti.existsAP(_ssid) == false) {
271  _wifiMulti.addAP(_ssid, _pass);
272  }
273

275  WS_DEBUG_PRINTLN("CONNECTING");
276 
277  while (_wifiMulti.run(5000) != WL_CONNECTED &&
278  millis() - startRetry < 10000) {
279  // ESP8266 WDT requires yield() during a busy-loop so it doesn't bite
280  yield();
