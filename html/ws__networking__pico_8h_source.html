105  bool check_valid_ssid() {
106  // Set WiFi to station mode and disconnect from an AP if it was previously
107  // connected
108  WiFi.mode(WIFI_STA);
109  WiFi.disconnect();
110  delay(100);
111 
112  // Perform a network scan, CYW43 reports hidden networks with no SSID
113  int n = WiFi.scanNetworks();
114  if (n == 0) {
115  // a hidden network may still answer the directed scan below
116  WS_DEBUG_PRINTLN("No visible WiFi networks found!");
117  }
118 
119  bool foundNetwork = false;
120 
121  WS_DEBUG_PRINTLN("WipperSnapper found these WiFi networks:");
122  for (uint8_t i = 0; i < n; i++) {
123  const char *scanSsid = WiFi.SSID(i);
124  // hidden networks don't broadcast their SSID
125  bool hidden = (scanSsid == NULL || scanSsid[0] == '\0');
126  if (!hidden && !foundNetwork && strcmp(scanSsid, _ssid) == 0) {
127  foundNetwork = true;
128  } else if (!hidden && !foundNetwork && WS._isWiFiMulti) {
129  // multi network mode
130  for (int j = 0; j < WS_MAX_ALT_WIFI_NETWORKS; j++) {
131  if (strlen(WS._multiNetworks[j].ssid) > 0 &&
132  strcmp(WS._multiNetworks[j].ssid, scanSsid) == 0) {
133  foundNetwork = true;
134  }
135  }
136  }
137  if (hidden) {
138  WS_DEBUG_PRINT("<hidden>");
139  } else {
140  WS_DEBUG_PRINT(scanSsid);
141  }
142  WS_DEBUG_PRINT(" (");
143  uint8_t BSSID[WL_MAC_ADDR_LENGTH];
144  WiFi.BSSID(i, BSSID);
145  for (int m = 0; m < WL_MAC_ADDR_LENGTH; m++) {
146  if (m != 0)
147  WS_DEBUG_PRINT(":");
148  WS_DEBUG_PRINTHEX(BSSID[m]);
149  }
150  WS_DEBUG_PRINT(") ");
151  WS_DEBUG_PRINT(WiFi.RSSI(i));
152  WS_DEBUG_PRINT("dB (ch");
153  WS_DEBUG_PRINT(WiFi.channel(i))
154  WS_DEBUG_PRINTLN(")");
155  }
156 
157  // Hidden networks only reveal their SSID in reply to a directed probe.
158  // Not gated on seeing a hidden network above, CYW43 support is unverified.
159  _hiddenIdx = -1;
160  if (!foundNetwork) {
161  if (scan_hidden_ssid(_ssid)) {
162  _hiddenIdx = 0;
163  } else if (WS._isWiFiMulti) {
164  for (int j = 0; j < WS_MAX_ALT_WIFI_NETWORKS; j++) {
165  if (scan_hidden_ssid(WS._multiNetworks[j].ssid)) {
166  _hiddenIdx = j + 1;
167  break;
168  }
169  }
170  }
171  foundNetwork = (_hiddenIdx >= 0);
172  }
173 
174  if (!foundNetwork) {
175  WS_DEBUG_PRINTLN("ERROR: Your requested WiFi network was not found!");
176  }
177  return foundNetwork;
178  }

190  bool scan_hidden_ssid(const char *ssid) {
191 #if defined(PICO_CYW43_SUPPORTED)
192  if (ssid == NULL || strlen(ssid) == 0)
193  return false;
194 
195  WS_DEBUG_PRINT("Scanning for hidden WiFi network: ");
196  WS_DEBUG_PRINTLNVAR(ssid);
197  cyw43_wifi_scan_options_t scan_options;
198  memset(&scan_options, 0, sizeof(scan_options));
199  scan_options.ssid_len = strnlen(ssid, sizeof(scan_options.ssid));
200  memcpy(scan_options.ssid, ssid, scan_options.ssid_len);
201  _hiddenScan.ssid = ssid;
202  _hiddenScan.ssid_len = scan_options.ssid_len;
203  _hiddenScan.found = false;
204  WS.feedWDT();
205  if (cyw43_wifi_scan(&cyw43_state, &scan_options, &_hiddenScan,
206  _hiddenScanCB) != 0) {
207  return false;
208  }
209  uint32_t start = millis();
210  while (cyw43_wifi_scan_active(&cyw43_state) && millis() - start < 10000) {
211  // RP2040 WDT maxes out at ~8.3s, shorter than this wait
212  WS.feedWDT();
213  delay(10);
214  }
215  WS.feedWDT();
216  if (_hiddenScan.found) {
217  WS_DEBUG_PRINT("Found hidden WiFi network on ch");
218  WS_DEBUG_PRINTLNVAR(_hiddenScan.channel);
219  }
220  return _hiddenScan.found;
221 #else
222  (void)ssid;
223  return false;
224 #endif

408  void _connect() {
409 
410  if (WiFi.status() == WL_CONNECTED)
411  return;
412 
413  WiFi.mode(WIFI_STA);
414  WS.feedWDT();
415  WiFi.setTimeout(20000);
416  WS.feedWDT();
417 
418  if (strlen(_ssid) == 0) {
419  _status = WS_SSID_INVALID;
420  } else {
421  _disconnect();
422  delay(5000);
423  WS.feedWDT();
424  if (WS._isWiFiMulti && _hiddenIdx < 0) {
425  // multi network mode
426  _wifiMulti.clearAPList();
427  // add default network
428  _wifiMulti.addAP(_ssid, _pass);
429  // add array of alternative networks
430  for (int i = 0; i < WS_MAX_ALT_WIFI_NETWORKS; i++) {
431  // skip empty slots, they'd match hidden networks' empty SSIDs
432  if (strlen(WS._multiNetworks[i].ssid) > 0) {
433  _wifiMulti.addAP(WS._multiNetworks[i].ssid,
434  WS._multiNetworks[i].pass);
435  }
436  }
437  WS.feedWDT();
438  if (_wifiMulti.run(20000) == WL_CONNECTED) {
439  WS.feedWDT();
440  _status = WS_NET_CONNECTED;
441  return;
442  }
443  WS.feedWDT();
444  } else {
445  if (_hiddenIdx > 0) {
446  // hidden alternative network found by check_valid_ssid(), WiFiMulti
447  // can't see it
448  WiFi.begin(WS._multiNetworks[_hiddenIdx - 1].ssid,
449  WS._multiNetworks[_hiddenIdx - 1].pass);
450  } else {
451  // single network mode, or the hidden default network
452  WiFi.begin(_ssid, _pass);
453  }
454 
455  // Use the macro to retry the status check until connected / timed out
456  int lastResult;
457  RETRY_FUNCTION_UNTIL_TIMEOUT(
458  []() -> int { return WiFi.status(); }, // Function call each cycle
459  int, // return type
460  lastResult, // return variable (unused here)
461  [](int status) { return status == WL_CONNECTED; }, // check
462  PICO_CONNECT_TIMEOUT_MS, // timeout interval (ms)
463  PICO_CONNECT_RETRY_DELAY_MS); // interval between retries
464 
465  if (lastResult == WL_CONNECTED) {
466  _status = WS_NET_CONNECTED;
467  // wait 2seconds for connection to stabilize
468  WS_DELAY_WITH_WDT(2000);
469  return;
470  }
471  }
472  _status = WS_NET_DISCONNECTED;
473  }
474  }
